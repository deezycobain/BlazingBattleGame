function animateLunge(unitName,from,to,onImpact,onDone,attackKind='punch'){
 const token=ACTIVE_ACTION_TOKEN;
 let start=performance.now(),dur=175,backDur=145;
 let anim=ensureAnimState();
 if(!anim.attackPose)anim.attackPose={};
 anim.attackPose[unitName]={kind:attackKind,start,duration:dur+150};
 function go(now){
  if(!actionTokenAlive(token))return;
  let state=ensureAnimState();
  let t=Math.min(1,(now-start)/dur),ease=1-Math.pow(1-t,3);
  state.positions[unitName]={x:from.x+(to.x-from.x)*ease,y:from.y+(to.y-from.y)*ease};
  if(t<1)return requestAnimationFrame(go);
  try{onImpact&&onImpact()}catch(err){console.error('Attack impact callback failed:',err);return recoverAction('impact callback')}
  setTimeout(()=>{
   if(!actionTokenAlive(token))return;
   let bs=performance.now();
   function ret(n){
    if(!actionTokenAlive(token))return;
    let st=ensureAnimState();
    let q=Math.min(1,(n-bs)/backDur),e=1-Math.pow(1-q,3);
    st.positions[unitName]={x:to.x+(from.x-to.x)*e,y:to.y+(from.y-to.y)*e};
    if(q<1)return requestAnimationFrame(ret);
    delete st.positions[unitName];
    if(st.attackPose)delete st.attackPose[unitName];
    try{onDone&&onDone()}catch(err){console.error('Attack completion callback failed:',err);recoverAction('completion callback')}
   }
   requestAnimationFrame(ret)
  },65)
 }
 requestAnimationFrame(go)
}

// v0.6.7 DATA AUTHORITY
// Canonical stats/roles/shapes/ability costs live in each unit's data/unit.json.
// The standalone embeds a generated copy in window.BLAZING_UNIT_DATA so mobile local-file play needs no fetch().
const UNIT_DATA=window.BLAZING_UNIT_DATA||{};
const UNIT_NAME_TO_ID=Object.freeze(Object.fromEntries(Object.entries(UNIT_DATA).map(([id,d])=>[d.display_name,id])));
function canonicalUnit(nameOrId){
 const id=UNIT_DATA[nameOrId]?nameOrId:UNIT_NAME_TO_ID[nameOrId];
 const d=id?UNIT_DATA[id]:null;
 if(!d)throw new Error(`Unknown canonical unit: ${nameOrId}`);
 return d;
}
function runtimeDefinition(nameOrId){
 const d=canonicalUnit(nameOrId),basic=d.abilities.basic||{},jutsu=d.abilities.jutsu||{};
 return{
  mark:d.combat.mark,speed:d.stats.speed,attack:d.stats.attack,defense:d.stats.defense,
  hp:d.stats.hp,maxHp:d.stats.hp,shape:d.combat.basic_shape,jutsuShape:d.combat.jutsu_shape||{type:'circle',r:0},
  maxChakra:d.combat.chakra_max||0,jutsuCost:jutsu.cost??99,
  jutsuDamage:jutsu?window.BlazingCombatRuntime.computeScaledDamage(d.stats.attack,jutsu.damage_multiplier||1):0,
  jutsuName:jutsu?.name||'Jutsu',archetype:d.archetype,render:d.render||{},
  linkBuff:d.combat.link_buff||null
 };
}
const BATTLE_ROSTER=Object.freeze(Object.fromEntries(
 ['crimson','subzero','lebee','senku','anubis']
   .filter(id=>UNIT_DATA[id])
   .map(id=>[UNIT_DATA[id].display_name,runtimeDefinition(id)])
));
function makeRosterUnit(name,opts={}){
 const d=BATTLE_ROSTER[name];
 if(!d)throw new Error(`Unknown battle unit: ${name}`);
 const u=U(name,d.mark,d.speed,d.attack,d.shape,d.jutsuShape,d.jutsuCost,d.jutsuDamage,d.linkBuff||null,d.maxHp,d.maxChakra,d.defense,d.archetype);
 u.jutsuName=d.jutsuName||'Jutsu';
 u.renderScale=d.render?.scale||1;
 const start=opts.startingChakra??2;
 u.chakra=start==='full'?u.maxChakra:Math.max(0,Math.min(u.maxChakra,start));
 return u;
}
function makeEnemyFromData(id,name,x,y,mark,overrides={}){

function resolveEnemyOverlap(enemy){
 const minDist=47;

 // Resolve only the enemy that just moved. This intentionally does not
 // mutate player positions or other enemy positions.
 for(let pass=0;pass<8;pass++){
   let blocker=null;
   let blockerDist=Infinity;

   S.pairs.forEach(pair=>{
     if(!pairAlive(pair))return;
     let dist=d(enemy,pair);
     if(dist<minDist && dist<blockerDist){
       blocker=pair;
       blockerDist=dist;
     }
   });

   S.enemies.forEach(other=>{
     if(other===enemy||other.hp<=0)return;
     let dist=d(enemy,other);
     if(dist<minDist && dist<blockerDist){
       blocker=other;
       blockerDist=dist;
     }
   });

   if(!blocker)break;

   let dx=enemy.x-blocker.x;
   let dy=enemy.y-blocker.y;
   let len=Math.hypot(dx,dy);

   // Exact same-center fallback gets a deterministic escape direction.
   if(len<.01){
     let angle=(pass+1)*1.913;
     dx=Math.cos(angle);
     dy=Math.sin(angle);
     len=1;
   }

   let push=(minDist-len)+.75;
   enemy.x+=(dx/len)*push;
   enemy.y+=(dy/len)*push;

   let legal=clampToBattlefield(enemy);
   enemy.x=legal.x;
   enemy.y=legal.y;
 }
}
