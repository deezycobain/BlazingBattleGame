function animateLunge(unitName,from,to,onImpact,onDone,attackKind='punch'){
 const token=ACTIVE_ACTION_TOKEN;
 let start=performance.now(),dur=175,backDur=145;
 let anim=ensureAnimState();
 if(!anim.attackPose)anim.attackPose={};
 anim.attackPose[unitName]={kind:attackKind,start,duration:dur+150};
 function go(now){
  if(!actionTokenAlive(token))return;
  let state=ensureAnimState();
  let t=Math.min(1,(now-start)/dur),ease=1-Math.pow(1-t,3);
  state.positions[unitName]={x:from.x+(to.x-from.x)*ease,y:from.y+(to.y-from.y)*ease};
  if(t<1)return requestAnimationFrame(go);
  try{onImpact&&onImpact()}catch(err){console.error('Attack impact callback failed:',err);return recoverAction('impact callback')}
  setTimeout(()=>{
   if(!actionTokenAlive(token))return;
   let bs=performance.now();
   function ret(n){
    if(!actionTokenAlive(token))return;
    let st=ensureAnimState();
    let q=Math.min(1,(n-bs)/backDur),e=1-Math.pow(1-q,3);
    st.positions[unitName]={x:to.x+(from.x-to.x)*e,y:to.y+(from.y-to.y)*e};
    if(q<1)return requestAnimationFrame(ret);
    delete st.positions[unitName];
    if(st.attackPose)delete st.attackPose[unitName];
    try{onDone&&onDone()}catch(err){console.error('Attack completion callback failed:',err);recoverAction('completion callback')}
   }
   requestAnimationFrame(ret)
  },65)
 }
 requestAnimationFrame(go)
}

function drawField(){
  const img=S.bossBattle?ANUBIS_PORTRAIT_MAP:LEVEL1_PORTRAIT_MAP;
  if(img.complete && img.naturalWidth>0){
    ctx.imageSmoothingEnabled=true;
    // Portrait maps are authored for the phone battlefield: crop only if aspect ratios differ slightly.
    const ir=img.naturalWidth/img.naturalHeight, cr=W/H;
    let sx=0,sy=0,sw=img.naturalWidth,sh=img.naturalHeight;
    if(ir>cr){sw=img.naturalHeight*cr;sx=(img.naturalWidth-sw)/2;}
    else if(ir<cr){sh=img.naturalWidth/cr;sy=(img.naturalHeight-sh)/2;}
    // Camera zoom: retain the same battlefield coordinates, but crop the art
    // slightly tighter so combat reads closer on a phone.
    const baseSw=sw,baseSh=sh;
    sw=baseSw/MAP_ZOOM; sh=baseSh/MAP_ZOOM;
    sx+=(baseSw-sw)/2; sy+=(baseSh-sh)/2;
    ctx.drawImage(img,sx,sy,sw,sh,0,0,W,H);
    ctx.fillStyle='rgba(7,10,12,.025)';ctx.fillRect(0,0,W,H);
    return;
  }
  // Keep canvas transparent until the stage image is decoded; CSS background is the fallback.
  ctx.clearRect(0,0,W,H);
}

   pair.units.forEach(u=>{
     let pos=actorPos(pair,u);
     let target=nearestEnemyPoint(pos.x,pos.y);
     if(target)u.rotation=Math.atan2(target.y-pos.y,target.x-pos.x);
   });
 });

 S.enemies.forEach(e=>{
   let target=nearestPlayerPoint(e.x,e.y);
   if(target)e.rotation=Math.atan2(target.y-e.y,target.x-e.x);
 });
}
