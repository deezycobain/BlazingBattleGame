 // a proportionally larger footprint without turning into a detached blob.
 if(!isEnemy){
   let liftRatio=clamp(lift/17,0,1);
   let bodyScale=(sizeScale||1)*UNIT_RENDER_SCALE*1.07*PLAYER_RENDER_BONUS*(name==='Crimson'?CRIMSON_RENDER_BONUS:1);
   let shadowW=clamp(10.5*bodyScale,10,18.5);
   let shadowH=clamp(2.55*bodyScale,2.5,4.25);
   let shadowAlpha=.24-.08*liftRatio;

   ctx.save();
   ctx.translate(x,y+11.5);
   ctx.fillStyle=`rgba(0,0,0,${shadowAlpha})`;
   ctx.shadowColor=`rgba(0,0,0,${.14+.04*(1-liftRatio)})`;
   ctx.shadowBlur=1.5+4.5*liftRatio;
   ctx.beginPath();
   ctx.ellipse(0,0,shadowW,shadowH,0,0,Math.PI*2);
   ctx.fill();
   ctx.restore();
 }

 ctx.save();
 ctx.translate(x,y-lift);

 if(!visual){
   let t=(performance.now()+idlePhase*5);
   // Lebee's supplied idle sheet already contains floating motion.
   if(name!=='Lebee'){
     ctx.translate(0,Math.sin(t/420)*.55);
     ctx.rotate(Math.sin(t/610)*.004);
   }
 }

 if(fx){
   let elapsed=performance.now()-fx.start;
   if(elapsed<fx.duration){
     if(fx.kind==='hit'){
       let t=elapsed/fx.duration;
       let bump=Math.sin(t*Math.PI);
       ctx.translate((fx.dir||1)*7*bump,-1.6*bump);
       ctx.rotate((fx.dir||1)*0.14*bump*(fx.strong?1.15:1));
       scale*=1 + (fx.strong?.06:.035)*Math.sin(t*Math.PI*1.2);
     }else if(fx.kind==='ko'){
       let t=elapsed/fx.duration;
       let ease=1-Math.pow(1-t,3);
       ctx.translate((fx.dir||1)*(28*ease), -10*ease);
       ctx.rotate((fx.dir||1)*(.72*ease));
       scale*=1 - .18*ease;
       ctx.globalAlpha=t<.6 ? 1 : 1-((t-.6)/.4);
     }
   }else if(S.anim?.unitFX){
     delete S.anim.unitFX[name];
   }
 }

 ctx.rotate(rot);
 ctx.scale((flipX||1)*scale*activePulse,scale*activePulse);

 // CPU enemies retain their small orientation shadow.
 if(isEnemy){
   ctx.beginPath();
   ctx.ellipse(0,17,16,6,0,0,Math.PI*2);
   ctx.fillStyle='rgba(0,0,0,.22)';
   ctx.fill();
 }

 let sprite=name?SPRITES[name]:null;
 let idleFrames=name?unitIdleFrames(name):null;
 let attackState=(name&&S.anim?.attackPose)?S.anim.attackPose[name]:null;
 let attackFrames=(name&&attackState)?unitAttackFrames(name,attackState.kind):null;

 // Lebee has a real recoil sheet. Hit art temporarily overrides idle/attack art,
 // but uses the same normalized canvas so her footprint stays stable.
 if(!visual&&fx?.kind==='hit'&&HIT_SPRITES[name]?.length){
   const hitElapsed=performance.now()-fx.start;
   const hitProgress=clamp(hitElapsed/fx.duration,0,1);
   const hitIndex=Math.min(HIT_SPRITES[name].length-1,Math.floor(hitProgress*HIT_SPRITES[name].length));
   const hitSprite=HIT_SPRITES[name][hitIndex];
   if(hitSprite?.complete&&hitSprite.naturalWidth>0)sprite=hitSprite;
 }

 // Attack art overrides idle only when a REAL, loaded attack frame exists.
 // Previously every attack set attackState, which suppressed the idle art even
 // for units such as Deezy that have no dedicated attack frames. The renderer
 // then fell back to an older padded base sprite and the unit appeared to shrink.
 let usingAttackSprite=false;
 let attackProgress=0;
 if(!visual&&attackState){
   let elapsed=performance.now()-attackState.start;
   attackProgress=clamp(elapsed/attackState.duration,0,1);

   // Crimson's supplied sheet contains distinct idle / basic / Jutsu poses.
   // Briefly hold the second idle pose as anticipation, then snap into the
   // supplied action pose. This makes the frame change readable at phone scale
   // instead of showing one static pose for the entire lunge.
   if(name==='Crimson'&&idleFrames?.length){
     const anticipation=attackState.kind==='special'?.16:.22;
     if(attackProgress<anticipation){
       let prep=idleFrames[Math.min(1,idleFrames.length-1)];
       if(prep?.complete&&prep.naturalWidth>0){
         sprite=prep;
         usingAttackSprite=true;
       }
     }
   }

   if(!usingAttackSprite&&attackFrames?.length){
     let frameIndex=Math.min(attackFrames.length-1,Math.floor(attackProgress*attackFrames.length));
     let candidate=attackFrames[frameIndex];
     if(candidate?.complete&&candidate.naturalWidth>0){
       sprite=candidate;
       usingAttackSprite=true;
     }
   }
 }

 // Keep the normalized idle animation visible whenever no valid attack frame
 // is available. This guarantees a consistent battlefield footprint during
 // lunges and Jutsu casts instead of silently swapping to legacy sprite art.
 if(!usingAttackSprite && idleFrames?.length){
   const idleMs=name==='Lebee'?520:185;
   let frameIndex=Math.floor((performance.now()+idlePhase*7)/idleMs)%idleFrames.length;
   let candidate=idleFrames[frameIndex];
   if(candidate?.complete && candidate.naturalWidth>0)sprite=candidate;
 }

 let spriteRef=sprite?getRenderableSprite(sprite,name):null;
 let dims=spriteDims(spriteRef);

 if(spriteRef && dims.w>0 && dims.h>0){
   let playerScale=isEnemy?1:1.07;
   // v0.5.18: shared player scale; Crimson uses the same battlefield height target as Sub-Zero.
   let playerRenderBonus=isEnemy?1:PLAYER_RENDER_BONUS*(name==='Crimson'?CRIMSON_RENDER_BONUS:1);
   let h=(isEnemy?62:68)*(sizeScale||1)*UNIT_RENDER_SCALE*playerScale*playerRenderBonus,
       w=(isEnemy?42:48)*(sizeScale||1)*UNIT_RENDER_SCALE*playerScale*playerRenderBonus;
   let ratio=dims.w/dims.h;
   if(ratio>0)w=h*ratio;
   // Normalized action frames keep the exact same battlefield footprint as idle.

   ctx.imageSmoothingEnabled=true;

   // Deezy does not yet have a dedicated action-art pack. Give the existing
   // sprite a real attack performance rather than simply translating the whole
   // unit unchanged. Because this transform lives inside the already-facing
   // coordinate system, it automatically follows his attack direction.
   if(!visual&&attackState&&name==='Deezy'){
     let p=attackProgress;
     let strike=Math.sin(clamp(p/.78,0,1)*Math.PI);
     let recoil=p>.72?Math.sin(((p-.72)/.28)*Math.PI)*.35:0;
     let strength=attackState.kind==='special'?1.35:1;
     ctx.translate((8.5*strength*strike)-(2.5*recoil),-2.2*strike);
     ctx.rotate(-.105*strength*strike+.035*recoil);
     let squash=1+.055*strike;
     ctx.scale(squash,1-.028*strike);
   }

   if(active && !visual){
     let elapsed=Math.max(0,performance.now()-(S.readyStarted||0));
     let intro=Math.max(0,1-elapsed/230);
     let breathe=.5+.5*Math.sin(performance.now()/190);

     ctx.save();
     ctx.globalCompositeOperation='source-over';
     let crimson=name==='Crimson';
     ctx.shadowColor=crimson?'rgba(255,58,86,.98)':'rgba(255,255,255,.98)';
     ctx.shadowBlur=(crimson?9:6) + intro*10 + breathe*(crimson?4.5:2.5);
     ctx.globalAlpha=.32 + intro*.35 + breathe*.08;
     ctx.drawImage(spriteRef,-w/2,12-h,w,h);
     ctx.restore();

     if(intro>0){
       ctx.save();
       ctx.globalAlpha=.28*intro;
       ctx.globalCompositeOperation='screen';
       ctx.drawImage(spriteRef,-w/2,12-h,w,h);
       ctx.restore();
     }
   }

   ctx.globalAlpha=1;
   ctx.drawImage(spriteRef,-w/2,12-h,w,h);

 } else {
   ctx.beginPath();
   ctx.arc(0,0,16,0,Math.PI*2);
   ctx.fillStyle=color;
   ctx.fill();
   ctx.strokeStyle='rgba(255,255,255,.82)';
   ctx.lineWidth=1.3/Math.max(scale,0.01);
   ctx.stroke();

   ctx.fillStyle='#fff';
   ctx.font='bold 11px system-ui';
   ctx.textAlign='center';
   ctx.textBaseline='middle';
   ctx.fillText(label,0,0);
 }

 ctx.restore()
}

function playerLinkState(pair){
  const active=S.ready?.kind==='pair' ? S.ready.ref : null;
  if(!active || !pairAlive(active) || !pairAlive(pair))
    return {linked:false,count:0};

  let count=1;
  let linked=false;

  S.pairs.forEach(other=>{
    if(other===active || !pairAlive(other))return;
    if(unitsMutuallyLinked(active,other)){
      count++;
      if(pair===other)linked=true;
    }
  });

  if(pair===active && count>1)linked=true;
  return {linked,count};
}


function pairLinkVisualStrength(a,b){
 if(!a||!b||!unitsMutuallyLinked(a,b))return 0;

 // Mutual normal-link range is effectively normal radius + body radius.
 const outer=normalShape().r+16;
 const inner=30;
 const dist=d(a,b);
 const proximity=1-clamp((dist-inner)/Math.max(1,outer-inner),0,1);

 // If either fighter is actively providing a Link Attack Buff, let that
 // mechanical strength reinforce the visual strength too.
 const ab=linkBuffFromProvider(a,b).ratio||0;
 const ba=linkBuffFromProvider(b,a).ratio||0;

 return clamp(Math.max(proximity,ab,ba),0,1);
}

function linkVisualStrength(pair){
 const active=S.ready?.kind==='pair'?S.ready.ref:null;
 if(!active||!pairAlive(active)||!pairAlive(pair))return 0;

 if(pair===active){
   let best=0;
   S.pairs.forEach(other=>{
     if(other!==active&&pairAlive(other)&&unitsMutuallyLinked(active,other))
       best=Math.max(best,pairLinkVisualStrength(active,other));
   });
   return best;
 }

 return pairLinkVisualStrength(active,pair);
}

 const useDrag=isActive&&(S.drag||S.anim?.settle)&&S.dragVisual;
 const scale=useDrag?(S.dragVisual.scale||1):1;
 const lift=useDrag?(S.dragVisual.lift||0):0;

 // Player sprites are drawn 68px high with their bottom at local y=12.
 // This follows the real transformed top edge instead of using a fixed offset.
 return pair.y-lift-56*scale;
}

function drawActiveLinkSummary(pair,pos,spriteTop){
 if(S.phase!=='player'||S.ready?.kind!=='pair'||S.ready.ref!==pair)
   return;

 const previewing=(S.drag||S.action==='jutsu');
 if(!previewing)return;

 const members=comboMembers(null,pair);
 if(members.length<=1)return;

 const u=front(pair);
 let damage=0,baseDamage=0;

 if(S.action==='jutsu'){
   const helpers=members.filter(p=>p!==pair);
   damage=u.jutsuDamage+helpers.reduce((sum,p)=>sum+buffedNormalDamage(p).damage,0);
   baseDamage=u.jutsuDamage+helpers.reduce((sum,p)=>sum+front(p).attack,0);
 }else{
   damage=members.reduce((sum,p)=>sum+buffedNormalDamage(p).damage,0);
   baseDamage=members.reduce((sum,p)=>sum+front(p).attack,0);
 }

 const bonus=baseDamage>0?Math.max(0,damage/baseDamage-1):0;
 const strength=linkVisualStrength(pair);
 const sat=Math.round(28+72*strength);
 const alpha=.45+.50*strength;
 const accent=`hsla(143,${sat}%,58%,${alpha})`;

 const label=`LINK ×${members.length} · ${damage} DMG${bonus>0?` · +${Math.round(bonus*100)}%`:''}`;
 const y=spriteTop-27;

 ctx.save();
 ctx.font='bold 9px system-ui';
 ctx.textAlign='center';
 const width=Math.max(84,ctx.measureText(label).width+12);

 ctx.fillStyle='rgba(7,10,12,.68)';
 ctx.fillRect(pos.x-width/2,y-12,width,17);

 ctx.strokeStyle=accent;
 ctx.lineWidth=.8;
 ctx.strokeRect(pos.x-width/2+.5,y-11.5,width-1,16);

 ctx.fillStyle=accent;
 ctx.fillText(label,pos.x,y);
 ctx.restore();
}

function render(){
 try{
 // Start every frame from a known-good canvas state.
 ctx.globalAlpha=1;
 ctx.globalCompositeOperation='source-over';
 ctx.shadowBlur=0;
 updateFacing();
 drawField();
 if(S.jutsuDim){
   const now=performance.now();
   const elapsed=now-S.jutsuDim.start;
   const fadeIn=clamp(elapsed/220,0,1);
   const fadeOut=S.jutsuDim.end?clamp((now-S.jutsuDim.end)/260,0,1):0;
   const a=(S.jutsuDim.alpha||.42)*fadeIn*(1-fadeOut);
   if(a>0){ctx.save();ctx.fillStyle=`rgba(3,2,10,${a})`;ctx.fillRect(0,0,W,H);ctx.restore()}
   if(S.jutsuDim.end&&fadeOut>=1)S.jutsuDim=null;
 }
 drawMoveReturnCue();

 const showPlayer=S.phase==='player'&&S.ready?.kind==='pair'&&(S.drag||S.action==='normal'||S.action==='jutsu');
 const activePair=S.ready?.kind==='pair'?S.ready.ref:null;

 // Attack/range fields live on the ground, so draw them before the pieces.
 if(showPlayer&&activePair&&pairAlive(activePair)){
   ensureFront(activePair);
   const activeUnit=front(activePair);
   const activePos=actorPos(activePair,activeUnit);
   const proxy=attackProxy(activeUnit);
   const jutsu=S.action==='jutsu';
   const attackColor=!jutsu
     ? '#ffffff'
     : (isUltimateAction(activeUnit)?'#ff3548':'#45d7ff');
   drawShape(activePos,proxy,attackColor,jutsu?.22:.12,jutsu,-22);
 }

 S.enemies.forEach(e=>{
   if(e.hp<=0 && !renderWhileKO(e.name))return;
   const pos=(S.anim&&S.anim.positions&&S.anim.positions[e.name])||e;
   if(S.anim?.showEnemyRange===e)
     drawShape(pos,{shape:normalShape(),rotation:0},'#ffffff',.14,false);
 });

 // Build ONE scene list so team identity never controls visual stacking.
 const scene=[];

 S.pairs.forEach((p,i)=>{
   if(!pairRenderable(p))return;
   ensureFront(p);

   const u=front(p);
   const pos=actorPos(p,u);
   const selected=activePair===p;
   const lifted=selected&&((S.drag&&S.dragVisual)||(S.anim?.settle&&S.dragVisual));

   scene.push({
     kind:'player',
     y:pos.y,
     sortY:pos.y+(lifted?10000:0),
     pair:p,index:i,unit:u,pos,selected,lifted
   });
 });

 S.enemies.forEach(e=>{
   if(e.hp<=0 && !renderWhileKO(e.name))return;
   const pos=(S.anim&&S.anim.positions&&S.anim.positions[e.name])||e;
   scene.push({
     kind:'enemy',
     y:pos.y,
     sortY:pos.y,
     enemy:e,pos
   });
 });

 // Higher on map = farther away = rendered first.
 // Lower on map = foreground = rendered later/on top.
 scene.sort((a,b)=>{
   if(a.sortY!==b.sortY)return a.sortY-b.sortY;
   // Stable deterministic tie break when two baselines are essentially equal.
   return a.kind==='enemy'&&b.kind==='player'?-1:
          a.kind==='player'&&b.kind==='enemy'?1:0;
 });

 // BODY PASS — depth sorted.
 scene.forEach(item=>{
   if(item.kind==='player'){
     const {pair:p,index:i,unit:u,pos,selected}=item;
     const vis=((S.drag||S.anim?.settle)&&selected)?S.dragVisual:null;

     ctx.save();
     ctx.globalCompositeOperation='source-over';
     ctx.shadowBlur=0;
     ctx.globalAlpha=(activePair && !selected) ? .82 : 1;
     drawUnit(
       pos.x,pos.y,u.mark,COLORS[i],
       S.phase==='player'&&selected,
       vis,u.name,false,facingFlip(u.rotation,u.name),(u.renderScale||1)
     );
     ctx.restore();
     return;
   }

   const e=item.enemy;
   const pos=item.pos;
   const activePlayer=S.ready?.kind==='pair'?front(S.ready.ref):null;
   const proxy=activePlayer?attackProxy(activePlayer):null;
   const supportJutsu=activePlayer&&S.action==='jutsu'&&isAllySupportJutsu(activePlayer);
   const liveHit=showPlayer&&activePair&&!supportJutsu&&hits(activePair,proxy,e);

   // Target feedback is part of the enemy's world layer.
   if(liveHit){
     const jutsu=S.action==='jutsu';

     const linkedCount=comboMembers(e,activePair).length;
     const comboLinked=linkedCount>1;
     const targetColor=!jutsu
       ? (comboLinked?'#39ff88':'#ffffff')
       : (isUltimateAction(activePlayer)?'#ff3548':'#45d7ff');

     // Bubble feedback is isolated so it can never mask the enemy sprite.
     ctx.save();
     ctx.globalCompositeOperation='source-over';
     ctx.globalAlpha=1;
     ctx.shadowBlur=0;

     ctx.beginPath();
     ctx.arc(pos.x,pos.y,(e.r||19)+14,0,Math.PI*2);
     ctx.fillStyle=!jutsu
       ? 'rgba(255,255,255,.08)'
       : (isUltimateAction(activePlayer)?'rgba(255,53,72,.09)':'rgba(69,215,255,.08)');
     ctx.fill();

     ctx.beginPath();
     ctx.arc(pos.x,pos.y,(e.r||19)+10+1.5*Math.sin(performance.now()/145),0,Math.PI*2);
     ctx.strokeStyle=targetColor;
     ctx.shadowColor=targetColor;
     ctx.shadowBlur=jutsu?10:(comboLinked?8:6);
     ctx.lineWidth=jutsu?1.8:1.45;
     ctx.stroke();
     ctx.restore();
   }

   ctx.save();
   ctx.globalAlpha=1;
   drawUnit(
     pos.x,pos.y,e.mark,'#d84b54',
     S.phase==='cpu'&&S.ready?.ref===e,
     null,(e.spriteKey||e.name),true,facingFlip(e.rotation,(e.spriteKey||e.name)),(e.sizeScale||1)
   );
   ctx.restore();
 });

 // HUD PASS — stays readable above the depth-sorted bodies.
 const supportPreviewUnit=(S.ready?.kind==='pair')?front(S.ready.ref):null;
 const showAllyHealTargets=S.phase==='player'&&S.action==='jutsu'&&supportPreviewUnit&&isAllySupportJutsu(supportPreviewUnit);

 S.pairs.forEach(p=>{
   if(!pairRenderable(p))return;
   ensureFront(p);

   const u=front(p);
   if(u.hp<=0)return;
   const pos=actorPos(p,u);
   const selected=activePair===p;

   if(showAllyHealTargets){
     const pulse=.5+.5*Math.sin(performance.now()/170);
     ctx.save();
     ctx.globalCompositeOperation='source-over';
     ctx.beginPath();
     ctx.arc(pos.x,pos.y-15,31+3*pulse,0,Math.PI*2);
     ctx.fillStyle='rgba(86,255,122,.10)';
     ctx.fill();
     ctx.strokeStyle='rgba(108,255,139,.92)';
     ctx.shadowColor='#5dff76';
     ctx.shadowBlur=10+4*pulse;
     ctx.lineWidth=1.6;
     ctx.stroke();

     // Healing cross above each allied unit.
     ctx.translate(pos.x,pos.y-58);
     ctx.fillStyle='rgba(126,255,151,.96)';
     ctx.shadowColor='#69ff82';
     ctx.shadowBlur=8;
     ctx.fillRect(-2.5,-8,5,16);
     ctx.fillRect(-8,-2.5,16,5);
     ctx.restore();
   }

   const linkState=playerLinkState(p);
   drawPlayerResources(pos,u,linkState,selected,p);
 });

 S.enemies.forEach(e=>{
   if(e.hp<=0)return;
   if(renderWhileKO(e.name))return;
   const pos=(S.anim&&S.anim.positions&&S.anim.positions[e.name])||e;

   ctx.save();
   ctx.globalAlpha=S.phase==='player'?.92:1;
   ctx.fillStyle='#222';
   ctx.fillRect(pos.x-27,pos.y+29,54,6);
   ctx.fillStyle='#ffbd4a';
   ctx.fillRect(pos.x-27,pos.y+29,54*(e.hp/e.maxHp),6);
   ctx.restore();
 });

 S.floaters.forEach(f=>{
   ctx.save();
   ctx.globalAlpha=f.life??1;
   if(f.kind==='screenFlash'){
     const elapsed=performance.now()-f.start;
     const t=clamp(elapsed/f.duration,0,1);
     ctx.setTransform(1,0,0,1,0,0);
     ctx.globalCompositeOperation='source-over';
     ctx.fillStyle=f.color||'#fff';
     ctx.globalAlpha=f.fade?Math.max(0,1-t):1;
     ctx.fillRect(0,0,W,H);
   }else if(f.kind==='lebeeStarProjectile'){window.BlazingVfxRenderer.drawLebeeStarProjectile(ctx,f,LEBEE_STAR_PROJECTILE);}else if(f.kind==='senkuChemProjectile'){
     const t=clamp((performance.now()-f.start)/f.duration,0,1);
     const e=1-Math.pow(1-t,2.15);
     const x=f.from.x+(f.to.x-f.from.x)*e;
     const y=f.from.y+(f.to.y-f.from.y)*e-7*Math.sin(t*Math.PI);
     const idx=Math.min(SENKU_CHEM_PROJECTILE_FRAMES.length-1,Math.floor(t*SENKU_CHEM_PROJECTILE_FRAMES.length));
     const img=SENKU_CHEM_PROJECTILE_FRAMES[idx];
     if(img?.complete&&img.naturalWidth>0){
       const h=30+5*Math.sin(t*Math.PI),ratio=img.naturalWidth/img.naturalHeight,w=h*ratio;
       ctx.translate(x,y);
       ctx.shadowColor='#62ff63';ctx.shadowBlur=10;
       ctx.drawImage(img,-w/2,-h/2,w,h);
     }
   }else if(f.kind==='senkuChemImpact'){
     const t=clamp((performance.now()-f.start)/f.duration,0,1);
     const idx=Math.min(SENKU_CHEM_IMPACT_FRAMES.length-1,Math.floor(t*SENKU_CHEM_IMPACT_FRAMES.length));
     const img=SENKU_CHEM_IMPACT_FRAMES[idx];
     if(img?.complete&&img.naturalWidth>0){
       const h=72+45*Math.sin(Math.min(1,t)*Math.PI*.72);

function linkBuffFromProvider(providerPair,receiverPair){
 if(!pairAlive(providerPair)||!pairAlive(receiverPair)||providerPair===receiverPair)
   return {bonus:0,ratio:0,provider:null};

 ensureFront(providerPair);ensureFront(receiverPair);
 let provider=front(providerPair);
 if(!provider.linkBuff)return {bonus:0,ratio:0,provider:null};

 // Buff only matters when the two units are currently combo-linked.
 if(!unitsMutuallyLinked(providerPair,receiverPair))
   return {bonus:0,ratio:0,provider:null};

 let dist=d(providerPair,receiverPair);
 let range=provider.linkBuff.range;
 if(dist>range)return {bonus:0,ratio:0,provider:null};

 // A small "full power" distance keeps the curve from exploding at zero.
 let inner=34;
 let t=1-clamp((dist-inner)/Math.max(1,range-inner),0,1);
 let bonus=provider.linkBuff.minBonus+
           (provider.linkBuff.maxBonus-provider.linkBuff.minBonus)*t;

 return {bonus,ratio:t,provider};
}

function totalLinkAttackBuff(receiverPair){
 let total=0,bestRatio=0,providers=[];
 S.pairs.forEach(providerPair=>{
   if(providerPair===receiverPair)return;
   let result=linkBuffFromProvider(providerPair,receiverPair);
   if(result.bonus>0){
     total+=result.bonus;
     bestRatio=Math.max(bestRatio,result.ratio);
     providers.push(result.provider.name);
   }
 });
 return {bonus:total,ratio:bestRatio,providers};
}
